-- 2. MÓDULO SISTEMA, SEGURIDAD Y CONFIGURACIÓN GLOBAL
-- =================================================================

CREATE TABLE "Sist_Roles" (
    "Id_Rol" INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    "Nombre" TEXT NOT NULL,
    "Descripcion" TEXT,
    "Activo" BOOLEAN DEFAULT TRUE,
    "Es_Predeterminado" BOOLEAN DEFAULT FALSE,
    "Icono" TEXT DEFAULT 'fa-user-shield',
    "Fecha_Creacion" TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE "Sist_Modulos" (
    "Id_Modulo" INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    "Nombre_Clave" TEXT NOT NULL,
    "Nombre_Visible" TEXT NOT NULL,
    "Orden_Visual" INT DEFAULT 0,
    "VisibleRoles" BOOLEAN DEFAULT TRUE,
    "Activo" BOOLEAN DEFAULT TRUE,
    "Fecha_Creacion" TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE "Sist_Usuarios" (
    "Id_Usuario" INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    "NombreCompleto" TEXT NOT NULL,
    "Nombre_Usuario" TEXT UNIQUE,
    "Email" TEXT NOT NULL UNIQUE,
    "Email_Temporal" VARCHAR(255),
    "PasswordHash" TEXT NOT NULL,
    "Telefono" TEXT,
    "Id_Rol" INT REFERENCES "Sist_Roles"("Id_Rol"),
    "Activo" BOOLEAN DEFAULT TRUE,
    "Email_Verificado" BOOLEAN DEFAULT FALSE,
    "Sello_Seguridad" UUID DEFAULT gen_random_uuid(),
    "Fecha_Nacimiento" DATE,
    "Genero" VARCHAR(20),
    "Id_Iglesia_Asignada" INTEGER NULL,
    "FechaRegistro" TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    "Fecha_Modificacion" TIMESTAMP
);

CREATE TABLE "Sist_Permisos" (
    "Id_Permiso" INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    "Id_Rol" INT REFERENCES "Sist_Roles"("Id_Rol"),
    "Id_Modulo" INT REFERENCES "Sist_Modulos"("Id_Modulo"),
    "P_Leer" BOOLEAN DEFAULT FALSE,
    "P_Crear" BOOLEAN DEFAULT FALSE,
    "P_Editar" BOOLEAN DEFAULT FALSE,
    "P_Borrar" BOOLEAN DEFAULT FALSE,
    "P_Admin" BOOLEAN DEFAULT FALSE
);

CREATE TABLE "Sist_Verificaciones" (
    "Id_Verificacion" INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    "Id_Usuario" INT REFERENCES "Sist_Usuarios"("Id_Usuario") ON DELETE CASCADE,
    "Token" TEXT NOT NULL,
    "Fecha_Expiracion" TIMESTAMP NOT NULL,
    "Fecha_Creacion" TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE "Sist_Bitacora" (
    "Id_Bitacora" SERIAL PRIMARY KEY,
    "Fecha" TIMESTAMP DEFAULT NOW(),
    "Id_Usuario" INT,
    "Modulo" TEXT,
    "Accion" TEXT,
    "Detalle" TEXT,
    "IP" TEXT
);

CREATE TABLE "Sist_Visitas" (
    "Id_Visita" SERIAL PRIMARY KEY,
    "Fecha" TIMESTAMP DEFAULT NOW(),
    "Ip_Address" VARCHAR(50),
    "Url_Visitada" VARCHAR(255),
    "User_Agent" TEXT,
    "Dispositivo" VARCHAR(50),
    "Id_Usuario" INT DEFAULT 0
);

CREATE TABLE "Sist_Parametros" (
    "Clave" VARCHAR(50) PRIMARY KEY,
    "Valor" TEXT NOT NULL,
    "Descripcion" TEXT,
    "Activo" BOOLEAN DEFAULT TRUE
);

CREATE TABLE "Sist_EnvioCorreos" (
    "Id_Alerta" SERIAL PRIMARY KEY,
    "Clave_Evento" VARCHAR(100) UNIQUE NOT NULL,
    "Descripcion" TEXT,
    "Correos_Destino" TEXT NOT NULL,
    "Activo" BOOLEAN DEFAULT TRUE
);

CREATE TABLE "Sist_CodigosQR" (
    "Id_QR" SERIAL PRIMARY KEY,
    "Titulo" VARCHAR(150) NOT NULL,
    "Url" TEXT NOT NULL,
    "ColorFrente" VARCHAR(10) DEFAULT '#000000',
    "ColorFondo" VARCHAR(10) DEFAULT '#FFFFFF',
    "FormaOjos" VARCHAR(50) DEFAULT 'Cuadrado',
    "FormaModulos" VARCHAR(50) DEFAULT 'Cuadrado',
    "ImagenBase64" TEXT NOT NULL,
    "Id_Usuario" INTEGER NOT NULL,
    "Fecha_Creacion" TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE "Fin_Caja" (
    "Id_Movimiento" SERIAL PRIMARY KEY,
    "Concepto" TEXT NOT NULL,
    "Monto" DECIMAL(12,2) NOT NULL,
    "Tipo" TEXT CHECK ("Tipo" IN ('Ingreso', 'Egreso')),
    "Fecha" TIMESTAMP DEFAULT NOW(),
    "Mes" INT GENERATED ALWAYS AS (EXTRACT(MONTH FROM "Fecha")) STORED,
    "Anio" INT GENERATED ALWAYS AS (EXTRACT(YEAR FROM "Fecha")) STORED,
    "Id_Usuario" INT REFERENCES "Sist_Usuarios"("Id_Usuario"),
    "Id_Archivo" INT REFERENCES "Rec_Archivos"("Id_Archivo"),
    "Movimiento_En_Banco" BOOLEAN DEFAULT FALSE,
    "Editado" BOOLEAN NOT NULL DEFAULT FALSE,
    "Folio_Bancario" VARCHAR(100),
    "Id_Evento" INTEGER NULL
);

CREATE TABLE "Fin_Movimientos" (
    "Id_Movimiento" INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    "Monto" DECIMAL(18,2) NOT NULL,
    "Tipo" TEXT NOT NULL,
    "Concepto" TEXT NOT NULL,
    "Fecha_Movimiento" TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    "Id_Usuario_Registra" INT REFERENCES "Sist_Usuarios"("Id_Usuario"),
    "Fecha_Creacion" TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE "Fin_Evidencias" (
    "Id_Evidencia" INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    "Id_Movimiento" INT REFERENCES "Fin_Movimientos"("Id_Movimiento"),
    "Nombre_Archivo" TEXT,
    "Extension" TEXT,
    "Contenido_Binario" BYTEA,
    "Fecha_Creacion" TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE "Fin_Traspasos" (
    "Id_Traspaso" SERIAL PRIMARY KEY,
    "Id_Usuario_Origen" INTEGER NOT NULL REFERENCES "Sist_Usuarios"("Id_Usuario"),
    "Id_Usuario_Destino" INTEGER NOT NULL REFERENCES "Sist_Usuarios"("Id_Usuario"),
    "Monto" DECIMAL(18,2) NOT NULL,
    "Estado" VARCHAR(3) NOT NULL DEFAULT 'PEN' CHECK ("Estado" IN ('PEN', 'ACE', 'REC')),
    "Concepto" VARCHAR(255) NOT NULL,
    "Fecha_Solicitud" TIMESTAMP NOT NULL DEFAULT NOW(),
